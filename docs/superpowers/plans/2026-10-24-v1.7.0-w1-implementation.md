# v1.7.0 W1 — reliability and the independent kit: implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A cancelled call never shares the document with its still-running worker;
every QA check reads every entity of every space and reports a failure instead of
a clean result; no exported text becomes a spreadsheet formula; spec inputs are
validated in one place; the premium skill is back as a Claude Code plugin;
search-mode clients can read one tool's full schema; the standards coverage is
published.

**Architecture:** Lifecycle fixes inside `_async` / `_run` and the quarantine
module; one shared all-spaces snapshot per QA run; two small pure modules
(`engineering/export/cells.py`, `engineering/validation.py`); a plugin directory
generated from the canonical skill; one synthetic discovery tool. No new contract
member.

**Tech Stack:** Python 3.11, FastMCP 3.4.5, ezdxf 1.4.4, pywin32 (COM), openpyxl,
anyio, pytest + pytest-asyncio, ruff — all through `uv run --frozen`.

**Spec:** `docs/superpowers/specs/2026-10-24-v1.7.0-design.md` (§3 execution
model, §4 W1, §8 refusals / Review Focus / gates).

```python
"""B01 — the document lease follows the worker thread, never the awaiting coroutine.

``EzdxfBackend._async`` released the document lock in ``except BaseException``
when the awaiting task was cancelled, but nothing can stop the worker thread:
a second call then ran against the same ``Drawing`` while the first worker was
still writing — with ``EZDXF_CALL_TIMEOUT`` 120 and 0 alike, and with no
quarantine. The MCP SDK cancels a request by cancelling an anyio scope, so a
client pressing Esc was enough.

Every test here is ordered by ``threading.Event``s, never by a sleep race: the
worker blocks on ``release`` until the test has observed what it needs to.
"""

from __future__ import annotations

import asyncio
import threading
import time

import pytest
import pytest_asyncio

import config
from backends.quarantine import AbandonedCall, DocumentQuarantineError

pytestmark = pytest.mark.asyncio


class _Gate:
    """A worker body that blocks until the test releases it."""

    def __init__(self, *, fail: bool = False):
        self.entered = threading.Event()
        self.release = threading.Event()
        self.finished = threading.Event()
        self.fail = fail

    def body(self, trace: list[str] | None = None):
        def _sync():
            self.entered.set()
            try:
                assert self.release.wait(10), "the test never released the worker"
                if trace is not None:
                    trace.append("A:end")
                if self.fail:
                    raise ValueError("died after the cancel")
                return "late"
            finally:
                self.finished.set()

        return _sync


async def _wait_for(event: threading.Event, timeout: float = 10.0) -> None:
    assert await asyncio.to_thread(event.wait, timeout), "event never set"


async def _until(predicate, timeout: float = 10.0) -> None:
    deadline = time.monotonic() + timeout
    while not predicate():
        if time.monotonic() > deadline:
            raise AssertionError(f"condition not met within {timeout:.1f}s")
        await asyncio.sleep(0.01)

# ── AbandonedCall.set_on_finish ─────────────────────────────────────────────


async def test_on_finish_runs_once_on_the_worker_thread():
    ran: list[str] = []
    watched = AbandonedCall(lambda: "value")
    assert watched.set_on_finish(lambda: ran.append(threading.current_thread().name)) is True
    worker = threading.Thread(target=watched, name="the-worker")
    worker.start()
    worker.join(5)
    assert ran == ["the-worker"]
    assert watched.finished is True


async def test_on_finish_refuses_after_the_worker_finished():
    watched = AbandonedCall(lambda: "value")
    watched()
    ran: list[int] = []
    assert watched.set_on_finish(lambda: ran.append(1)) is False
    assert ran == [], "a hook registered too late is the caller's job, never run twice"


# ── cancelled while queued ──────────────────────────────────────────────────


@pytest.mark.parametrize("timeout", [30.0, 0.0])
async def test_cancel_while_queued_touches_no_lock(backend, monkeypatch, timeout):
    monkeypatch.setattr(config.settings, "ezdxf_call_timeout", timeout)
    ran: list[str] = []
    held = backend._lock
    await held.acquire()  # a live call owns the document
    try:
        task = asyncio.ensure_future(backend._async(lambda: ran.append("queued")))
        for _ in range(5):
            await asyncio.sleep(0)
        task.cancel()
        with pytest.raises(asyncio.CancelledError):
            await task
        assert held.locked(), "the queued call must not release a lock it never took"
        assert ran == []
    finally:
        held.release()
    assert backend.call_outcomes[-1]["outcome"] == "cancelled_queued"
    assert await backend._async(lambda: "next") == "next"
```

```python
class AbandonedCall:
    """Observation post for a worker thread ``_async`` can no longer await.

    The obvious hook does not work, and this was checked before being relied on:
    ``asyncio.timeout_at`` **cancels** the Future ``_async`` awaits, so by the
    time the abandon branch runs that Future is already done and
    ``add_done_callback`` fires immediately with ``cancelled()`` True. It reports
    the *await* ending, never the thread — which nothing can cancel. The only
    place the thread's outcome is observable is inside the thread, so
    ``_async`` wraps every call in one of these.

    v1.7.0 (B01): the document lease follows the worker thread, not the
    awaiting coroutine. When the coroutine is cancelled while the worker runs,
    ``_async`` registers a one-shot hook with ``set_on_finish`` and the worker
    calls it as it ends. ``_hook_lock`` closes the race with the worker
    finishing at that moment: either ``set_on_finish`` sees ``finished`` and
    returns False (the caller releases at once), or the worker takes the hook
    and runs it — never both, never neither. ``finished`` stays a plain bool for
    the readers that only poll it.
    """

    __slots__ = ("func", "finished", "result", "error", "_hook", "_hook_lock")

    def __init__(self, func):
        self.func = func
        self.finished = False
        self.result: Any = None
        self.error: BaseException | None = None
        self._hook: Callable[[], None] | None = None
        self._hook_lock = threading.Lock()

    def set_on_finish(self, hook: Callable[[], None]) -> bool:
        """Register ``hook`` for the worker thread to call when it ends.

        Returns False when the worker has already finished: the hook is not
        stored and the caller does the hook's work itself, now. A registered
        hook runs exactly once, on the worker thread.
        """
        with self._hook_lock:
            if self.finished:
                return False
            self._hook = hook
            return True

    def __call__(self, *args, **kwargs):
        try:
            self.result = self.func(*args, **kwargs)
            return self.result
        except BaseException as exc:
            self.error = exc
            raise
        finally:
            with self._hook_lock:
                self.finished = True
                hook, self._hook = self._hook, None
            if hook is not None:
                try:
                    hook()
                except Exception:  # a hook failure must not replace the call's outcome
                    log.exception("on_finish hook of %r failed", real_call_name(self.func))


class DaemonExecutor(concurrent.futures.Executor):
    """Worker threads that never hold the interpreter open.

    ``asyncio.to_thread`` runs on the loop's default ``ThreadPoolExecutor``,
    whose threads ``asyncio.run`` (``shutdown_default_executor``) and
    ``concurrent.futures``' exit hook both join. A worker nothing can cancel
    therefore kept the server process alive until it finished — measured on
    2026-10-24: a 6 s runaway held ``asyncio.run`` for 6.0 s and the process for
    7.2 s after the call had been abandoned. These threads are daemon threads
    that nothing joins at exit; ``disconnect()`` waits a bounded
    ``SHUTDOWN_GRACE`` for them instead.

    Reuse follows ``ThreadPoolExecutor`` (CPython 3.13): a worker marks itself
    idle only when the queue is empty, and ``submit`` claims an idle worker
    before spawning a new one. ``max_workers=1`` keeps one strictly FIFO thread
    (the COM STA apartment).
    """

    def __init__(
        self,
        max_workers: int | None = None,
        *,
        name: str = "acadmcp-worker",
        initializer: Callable[[], None] | None = None,
    ):
        if max_workers is not None and max_workers < 1:
            raise ValueError("max_workers must be at least 1")
        self._max_workers = max_workers
        self._name = name
        self._initializer = initializer
        self._queue: queue.SimpleQueue = queue.SimpleQueue()
        self._idle = threading.Semaphore(0)
        self._guard = threading.Lock()
        self._threads: list[threading.Thread] = []
        self._shutdown = False

    @property
    def threads(self) -> tuple[threading.Thread, ...]:
        return tuple(self._threads)

    def submit(self, fn, /, *args, **kwargs) -> concurrent.futures.Future:
        with self._guard:
            if self._shutdown:
                raise RuntimeError("cannot schedule new futures after shutdown")
            future: concurrent.futures.Future = concurrent.futures.Future()
            self._queue.put((future, fn, args, kwargs))
            if not self._idle.acquire(timeout=0) and (
                self._max_workers is None or len(self._threads) < self._max_workers
            ):
                thread = threading.Thread(
                    target=self._work, name=f"{self._name}-{len(self._threads)}", daemon=True
                )
                self._threads.append(thread)
                thread.start()
            return future

    def _work(self) -> None:
        if self._initializer is not None:
            try:
                self._initializer()
            except BaseException:
                log.exception("%s: worker initializer failed", self._name)
        while True:
            try:
                item = self._queue.get_nowait()
            except queue.Empty:
                self._idle.release()
                item = self._queue.get()
            if item is None:
                return
            future, fn, args, kwargs = item
            del item
            if future.set_running_or_notify_cancel():
                try:
                    result = fn(*args, **kwargs)
                except BaseException as exc:
                    future.set_exception(exc)
                else:
                    future.set_result(result)
                    del result
            del future, fn, args, kwargs

    def shutdown(self, wait: bool = True, *, cancel_futures: bool = False) -> None:
        with self._guard:
            self._shutdown = True
            threads = list(self._threads)
            if cancel_futures:
                while True:
                    try:
                        item = self._queue.get_nowait()
                    except queue.Empty:
                        break
                    if item is not None:
                        item[0].cancel()
        for _ in threads:
            self._queue.put(None)
        if wait:
            for thread in threads:
                if thread is not threading.current_thread():
                    thread.join()
```

```python
    def _pass_lease(self, lease: _Lease, watched: AbandonedCall) -> None:
        """Hand the document lock to the worker thread (B01).

        The awaiting coroutine is gone (cancelled), the worker is not: it keeps
        the lock until it ends, and the release reaches the loop through
        ``call_soon_threadsafe``. A loop that is already closed has nobody left
        to hand the lock to, so that case is dropped silently.
        """

            try:
                loop.call_soon_threadsafe(self._lease_finished, lease, watched)
            except RuntimeError:
                pass

        if not watched.set_on_finish(_on_finish):
            self._lease_finished(lease, watched)

    def _lease_finished(self, lease: _Lease, watched: AbandonedCall) -> None:
        """The worker holding a passed lease has ended (loop thread)."""
        if lease.timer is not None:
            lease.timer.cancel()
            lease.timer = None

        if lease.done:
            return

            # still queued on it were past the quarantine gate; releasing it
            # now would hand them the document behind the quarantine's back.
            return
        lease.lock.release()

        """The cancelled call's ORIGINAL deadline arrived (loop thread)."""
        lease.timer = None

            return  # finished in time; _lease_finished releases (or already did)

    async def _async(
        self, func, *args, quarantine_exit: bool = False, readonly: bool = False, **kwargs
    ):
        """Run a blocking ezdxf call in a worker thread, serialised and timed out.

        The ezdxf ``Drawing`` is not thread-safe, so every backend method funnels
        through one lock. ``EZDXF_CALL_TIMEOUT`` (seconds, ``0`` disables)
        bounds the *whole* wait — queueing behind an in-flight call plus the
        call itself — mirroring ``ComBackend._run``.

        **The lease follows the worker thread (v1.7.0, B01).** Nothing can stop
        a worker thread, so whatever ends the *wait* — a deadline, a
        cancellation (the MCP SDK cancels a request by cancelling an anyio
        scope, so a client pressing Esc is enough), a shutdown — the lock stays
        with the worker until it ends:

        - cancelled while queued: nothing was acquired, nothing is released;
        - cancelled while running: the cancelled coroutine does NOT release;
          ``AbandonedCall.set_on_finish`` hands the release to the worker,
          which posts it to the loop with ``call_soon_threadsafe`` exactly once.
          The call is recorded in ``call_outcomes``. If the worker is still
          alive at the call's ORIGINAL deadline, the abandon path below runs
          then (cause ``cancelled_then_deadline``);
        - deadline overrun: the worker's own future decides — ``work.done()``,
          never ``cancelled()``. Done means the call finished (or raised
          ``TimeoutError`` itself) as the deadline fired: its outcome is
          returned or raised and the lock released. Still running means the
          worker is abandoned: the lock is swapped, the orphaned one is never
          released (waiters already past the gate must not inherit it), and the
          document is quarantined;
        - ``readonly=True``: an overrun keeps the lease until the worker ends
          and later callers get "busy" — a read cannot corrupt the document, so
          it never quarantines it;
        - ``EZDXF_CALL_TIMEOUT=0``: no deadline, so nothing is ever abandoned or
          quarantined; a cancelled call's lease is still held until its worker
          ends.

        The document a call runs against (path and identity) is captured when
        it starts. Workers run on ``_WORKERS``, a ``DaemonExecutor``: a runaway
        can never hold the interpreter open at exit.

        R32 — every later call is refused **before** ``lock.acquire()`` while
        the document is quarantined (after the acquire the refusal would queue
        behind an unrelated call), and refused again after the acquire for a
        waiter that got past the gate before the quarantine began.
        ``quarantine_exit=True`` is for the three calls that rebind
        ``self._doc`` to a freshly constructed object; they still take the lock.
        """
        # The gate, before any await: a refusal must not queue.
        if self._quarantine is not None:
            self._reap_quarantine()
            if self._quarantine is not None and not quarantine_exit:
                raise quarantine_refusal(self._quarantine)

        timeout = config.settings.ezdxf_call_timeout
        loop = asyncio.get_running_loop()
        deadline = loop.time() + timeout if timeout > 0 else None
        lock = self._lock
        document_path = self._doc_path
        document_id = id(self._doc) if self._doc is not None else None
        document = document_path or self._active_key
        started = time.time()
        watched = AbandonedCall(func)
        acquired = False
        work: concurrent.futures.Future | None = None
        try:
            async with asyncio.timeout_at(deadline):
                await lock.acquire()
                acquired = True
                if self._quarantine is not None and not quarantine_exit:
                    self._reap_quarantine()
                    if self._quarantine is not None:
                        raise quarantine_refusal(self._quarantine)
                work = _WORKERS.submit(contextvars.copy_context().run, watched, *args, **kwargs)
                self._in_flight[work] = real_call_name(func)
                work.add_done_callback(self._forget_in_flight)
                result = await asyncio.wrap_future(work, loop=loop)
        except TimeoutError as exc:
            if not acquired:
                # Timed out queueing behind someone else's live call. That lock
                # is held by a call that will release it normally.
                raise self._busy_error(timeout) from exc
            if work is None:
                lock.release()
                raise
            if work.done():
                # Finished (or raised TimeoutError itself) as the deadline
                # fired: the worker is gone, so is the lease, and its outcome
                # is the answer. A never-started future is cancelled, not run.
                lock.release()
                if work.cancelled():
                    raise RuntimeError(
                        f"ezdxf call '{real_call_name(func)}' timed out after {timeout:g}s "
                        "before its worker started; nothing ran."
                    ) from exc
                return work.result()
            name = real_call_name(func)
            if readonly:
                self._pass_lease(_Lease(loop, lock, None, document_path, document_id), watched)
                log.warning(
                    "read-only ezdxf call %r exceeded EZDXF_CALL_TIMEOUT=%gs; it keeps the "
                    "document lock until it finishes and the document is not quarantined",
                    name,
                    timeout,
                )
                raise RuntimeError(
                    f"ezdxf read '{name}' timed out after {timeout:g}s. It is read-only, so "
                    "the document is NOT quarantined: its worker keeps the document lock until "
                    "it finishes and calls meanwhile answer 'busy'. Raise EZDXF_CALL_TIMEOUT "
                    "(seconds; 0 disables the timeout entirely) if this read is legitimately slow."
                ) from exc
            if self._lock is lock:
                self._lock = asyncio.Lock()
            self._quarantine_document(
                func, watched, timeout, document_path=document_path, document_id=document_id
            )
            self._record_outcome(func, "abandoned_deadline", started, document, ended=None)
            log.error(
                "ezdxf call %r exceeded EZDXF_CALL_TIMEOUT=%gs; abandoning the worker "
                "thread and the lock it holds, and quarantining the document it may "
                "still be writing to",
                name,
                timeout,
            )
            raise self._timeout_error(name, timeout) from exc
        except BaseException as exc:
            if not acquired:
                if isinstance(exc, asyncio.CancelledError):
                    self._record_outcome(
                        func, "cancelled_queued", started, document, ended=time.time()
                    )
                raise
            if work is None or work.cancel() or work.done():
                # Never started (cancelled before a worker took it), or already
                # ended: the lease ends here, with this coroutine.
                lock.release()
                if isinstance(exc, asyncio.CancelledError):
                    if work is None or work.cancelled():
                        outcome = "cancelled_queued"
                    elif work.exception() is not None:
                        outcome = "failed_after_cancel"
                    else:
                        outcome = "completed_after_cancel"
                    self._record_outcome(func, outcome, started, document, ended=time.time())
                raise
            # Cancelled while the worker runs: the lease passes to the worker.
            row = self._record_outcome(func, "cancelled_running", started, document, ended=None)
            lease = _Lease(loop, lock, row, document_path, document_id)
            if deadline is not None and not readonly:
                lease.timer = loop.call_at(
                    deadline, self._lease_deadline, lease, watched, func, timeout
                )
                self._leases.add(lease)
            self._pass_lease(lease, watched)
            raise
        lock.release()
        return result
```

```python
"""1.7.0 must not be slower than a base release: paired, alternating, calibrated.

Builds a ``git worktree`` of ``--base`` (a tag or commit), then measures every
workload on both trees in alternating order — base, head, head, base, ... — so
drift on the machine (thermal state, another process) lands on both sides
alike. Each measurement is one subprocess per tree that imports *that tree's*
packages (``PYTHONPATH`` = the tree, ``-P`` so this file's own directory is not
on the path) and runs the workloads of THIS checkout's ``perf_suite.py``, so
both trees run the same workload code.

Workloads: ``perf_suite``'s four (they drive the backend directly) plus
``server_create_lines_2k`` — 2,000 ``entity_create_line`` calls through an
in-memory FastMCP ``Client``, so the middleware and the tool layer are
measured too (about 7 s per measurement on the maintainer's machine).

``--aa`` adds an A/A calibration: the base tree is measured a second time in
every round, and the spread of base/base ratios is the noise floor. The gate
(spec §8.3): for every workload the median of the paired per-round ratios
head/base is at most ``1 + max(0.05, 2 × aa_spread)``. Without ``--aa`` the
spread is taken as 0 and the bound is 1.05; the phase gate runs with ``--aa``.

    python -m benchmarks.perf_compare --base v1.6.0 --aa --out out/perf/compare.json
"""

from __future__ import annotations

import argparse
import asyncio
import json
import os
import statistics
import subprocess
import sys
import tempfile
import time
from pathlib import Path

HERE = Path(__file__).resolve().parent
REPO = HERE.parent
PERF_SUITE = HERE / "perf_suite.py"

SUITE_WORKLOADS = ("create_lines_2k", "roundtrip_10k", "region_query_10k", "premium_pass")
SERVER_WORKLOAD = "server_create_lines_2k"
WORKLOADS = (*SUITE_WORKLOADS, SERVER_WORKLOAD)
MIN_MARGIN = 0.05


# ── the gate arithmetic (pure) ──────────────────────────────────────────────


def paired_ratio(numerators: list[float], denominators: list[float]) -> float:
    """Median of the per-round ratios ``numerators[i] / denominators[i]``."""
    if not numerators or len(numerators) != len(denominators):
        raise ValueError("paired_ratio needs two non-empty lists of the same length")
    return statistics.median(n / d for n, d in zip(numerators, denominators, strict=True))


def bound_for(aa_spread: float) -> float:
    """``1 + max(0.05, 2 × aa_spread)`` — the spec §8.3 ceiling on head/base."""
    return 1.0 + max(MIN_MARGIN, 2.0 * aa_spread)


def summarize(
    base: dict[str, list[float]],
    head: dict[str, list[float]],
    base_again: dict[str, list[float]] | None,
    *,
    base_ref: str,
    head_ref: str,
    warmup: int,
    rounds: int,
) -> dict:
    """The JSON report from per-round wall times in milliseconds."""
    aa_spread = 0.0
    if base_again is not None:
        aa_spread = max(abs(paired_ratio(base_again[w], base[w]) - 1.0) for w in base)
    bound = bound_for(aa_spread)
    workloads = {
        name: {
            "base_ms": round(statistics.median(base[name]), 3),
            "head_ms": round(statistics.median(head[name]), 3),
            "ratio": round(paired_ratio(head[name], base[name]), 4),
        }
        for name in base
    }
    return {
        "schema_version": 1,
        "base": base_ref,
        "head": head_ref,
        "warmup": warmup,
        "rounds": rounds,
        "aa": base_again is not None,
        "workloads": workloads,
        "aa_spread": round(aa_spread, 4),
        "bound": round(bound, 4),
        "ok": all(row["ratio"] <= bound for row in workloads.values()),
    }


# ── one measurement, inside a tree (the --worker side) ──────────────────────


def _load_suite():
    """This checkout's perf_suite, importing the *tree's* backends."""
    import importlib.util

    spec = importlib.util.spec_from_file_location("acadmcp_perf_suite", PERF_SUITE)
    module = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = module  # its dataclasses look their module up by name
    spec.loader.exec_module(module)
    return module


async def _server_create_lines_2k() -> float:
    os.environ["AUTOCAD_MCP_BACKEND"] = "ezdxf"
    import logging

    from fastmcp import Client

    import server

    logging.disable(logging.INFO)  # 4,000 audit lines would be measured too
    async with Client(server.mcp) as client:
        await client.call_tool("drawing_new", {})
        started = time.perf_counter()
        for index in range(2_000):
            y = float(index % 100)
            await client.call_tool(
                "entity_create_line", {"x1": 0.0, "y1": y, "x2": 100.0, "y2": y + 1.0}
            )
        return (time.perf_counter() - started) * 1000.0


async def measure(workloads: list[str], scratch: Path) -> dict[str, float]:
    """Wall milliseconds per workload, measured in this process."""
    suite = _load_suite()
    out: dict[str, float] = {}
    backend = suite.EzdxfBackend()
    await backend.connect()
    try:
        runners = {
            "create_lines_2k": lambda: suite.workload_create_lines_2k(backend),
            "roundtrip_10k": lambda: suite.workload_roundtrip_10k(backend, scratch),
            "region_query_10k": lambda: suite.workload_region_query_10k(backend),
            "premium_pass": lambda: suite.workload_premium_pass(backend),
        }
        for name in workloads:
            if name == SERVER_WORKLOAD:
                continue
            out[name] = float((await runners[name]()).wall_ms)
    finally:
        await backend.disconnect()
    if SERVER_WORKLOAD in workloads:
        out[SERVER_WORKLOAD] = await _server_create_lines_2k()
    return out


def run_in_tree(tree: Path, workloads: list[str], scratch: Path) -> dict[str, float]:
    """One measurement of ``workloads`` with ``tree``'s packages, in a subprocess."""
    env = {**os.environ, "PYTHONPATH": str(tree), "AUTOCAD_MCP_BACKEND": "ezdxf"}
    done = subprocess.run(
        [
            sys.executable,
            "-P",
            str(Path(__file__).resolve()),
            "--worker",
            "--workloads",
            ",".join(workloads),
            "--scratch",
            str(scratch),
        ],
        cwd=str(tree),
        env=env,
        capture_output=True,
        text=True,
        timeout=1800,
    )
    if done.returncode != 0:
        raise RuntimeError(f"measurement in {tree} failed:\n{done.stderr[-4000:]}")
    return json.loads(done.stdout.strip().splitlines()[-1])


# ── orchestration ───────────────────────────────────────────────────────────


def _git(*args: str) -> str:
    return subprocess.run(
        ["git", "-C", str(REPO), *args], check=True, capture_output=True, text=True
    ).stdout.strip()


def compare(
    base_ref: str, *, warmup: int, rounds: int, aa: bool, workloads: list[str], scratch: Path
) -> dict:
    head_ref = _git("rev-parse", "--short", "HEAD")
    tree = Path(tempfile.mkdtemp(prefix="acadmcp_perf_base_"))
    _git("worktree", "add", "--detach", "--quiet", str(tree), base_ref)
    samples: dict[str, dict[str, list[float]]] = {
        side: {w: [] for w in workloads} for side in ("base", "head", "base_again")
    }
    try:
        for index in range(warmup + rounds):
            order = [("base", tree), ("head", REPO)]
            if aa:
                order.append(("base_again", tree))
            if index % 2:
                order.reverse()
            measured = {side: run_in_tree(where, workloads, scratch) for side, where in order}
            if index < warmup:
                continue
            for side, values in measured.items():
                for name, value in values.items():
                    samples[side][name].append(value)
            print(f"round {index - warmup + 1}/{rounds} done", file=sys.stderr, flush=True)
    finally:
        subprocess.run(["git", "-C", str(REPO), "worktree", "remove", "--force", str(tree)])
    return summarize(
        samples["base"],
        samples["head"],
        samples["base_again"] if aa else None,
        base_ref=base_ref,
        head_ref=head_ref,
        warmup=warmup,
        rounds=rounds,
    )


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
    parser.add_argument("--base", default="v1.6.0", help="git ref to compare against")
    parser.add_argument("--warmup", type=int, default=5)
    parser.add_argument("--rounds", type=int, default=20)
    parser.add_argument("--aa", action="store_true", help="measure base twice per round")
    parser.add_argument("--out", type=Path, default=None)
    parser.add_argument("--workloads", default=",".join(WORKLOADS))
    parser.add_argument("--scratch", type=Path, default=Path("out/perf-scratch"))
    parser.add_argument("--worker", action="store_true", help=argparse.SUPPRESS)
    return parser


def main(argv: list[str] | None = None) -> int:
    args = build_parser().parse_args(argv)
    workloads = [w for w in args.workloads.split(",") if w]
    unknown = sorted(set(workloads) - set(WORKLOADS))
    if unknown:
        raise SystemExit(f"unknown workload(s) {unknown}; known: {list(WORKLOADS)}")
    args.scratch.mkdir(parents=True, exist_ok=True)
    if args.worker:
        print(json.dumps(asyncio.run(measure(workloads, args.scratch.resolve()))))
        return 0
    report = compare(
        args.base,
        warmup=args.warmup,
        rounds=args.rounds,
        aa=args.aa,
        workloads=workloads,
        scratch=args.scratch.resolve(),
    )
    rendered = json.dumps(report, indent=2, sort_keys=True)
    if args.out is not None:
        args.out.parent.mkdir(parents=True, exist_ok=True)
        args.out.write_text(rendered + "\n", encoding="utf-8")
    print(rendered)
    return 0 if report["ok"] else 1


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
"""One complete read of the drawing per QA run: every space, every entity, no cap.

``drawing_critique`` and the validator used to read ``entity_list(limit=5000)``
(10,000 for the validator) of the *current* space only, so a construction line
at position 5,001, a duplicate pair after the 5,000th line or anything on a
paper layout was never seen - and the result still read "zero issues". This
module is the one reader they share now:

* :func:`qa_snapshot` takes one all-spaces :class:`Snapshot` per run and caches
  it, with its :class:`ScanCoverage`, in ``shared["qa_snapshot"]``. A finalize
  passes one dict to the validator, the critique and the topology focuses, so
  it costs one read (one ``Document.Export`` on the live engine), not three.
  A dict is valid only while the drawing does not change: a caller that edits
  between two runs (``drawing_refine``) passes a fresh one.
* Before reading, the entities of every space are counted (``len`` of each
  ezdxf layout; ``Layout.Block.Count`` on the live engine). Above
  ``MAX_QA_ENTITIES`` (default 2,000,000; 0 disables) the read is refused and
  the coverage says why - a partial scan is never passed off as a whole one.
* Headless, the read runs under the backend's lock as a **read-only** call.
  ``_async(..., readonly=True)`` arrives with Task 1 on its own branch; until
  it has, :func:`_async_readonly` passes the flag only when ``_async``'s
  signature declares it, and never through ``**kwargs`` (which would hand
  ``readonly`` to the closure itself).
* Live, the active document is exported to a temporary DXF
  (``engineering/understand/snapshot.py::_com_export``) in the same COM job
  that counts its layouts, and the export's records per layout are compared
  with those live counts; any difference is an incomplete scan. Measured on
  AutoCAD 2026 on 2026-10-24 in a scratch document: per layout, ``Block.Count``
  equals the exported DXF's entity count - model space, an unused paper
  layout (0 and 0) and an activated layout holding its two VIEWPORTs and a
  LINE (3 and 3).

A read that fails for any reason other than a quarantine is cached as an
incomplete coverage carrying the reason, so every check of the run reports
``scan_incomplete`` once instead of retrying the read. A quarantine refusal
propagates: a quarantined document must never read as clean.
"""

from __future__ import annotations

import asyncio
import inspect
import shutil
from collections import Counter
from dataclasses import dataclass

import config
from backends.quarantine import DocumentQuarantineError
from engineering.understand.snapshot import Snapshot, _com_export, _read, records_from_doc

__all__ = [
    "QA_SNAPSHOT_KEY",
    "ScanCoverage",
    "entity_infos_all_spaces",
    "qa_snapshot",
    "shared_index",
]

#: The key :func:`qa_snapshot` caches its result under in a run's shared dict.
QA_SNAPSHOT_KEY = "qa_snapshot"


@dataclass(frozen=True)
class ScanCoverage:
    """What a QA read covered: ``complete`` only when every entity of every
    space was read."""

    complete: bool
    records: int
    spaces: tuple[str, ...]
    reason: str | None = None

    def to_dict(self) -> dict:
        return {
            "complete": self.complete,
            "records": self.records,
            "spaces": list(self.spaces),
            "reason": self.reason,
        }


def _accepts_readonly(backend) -> bool:
    """Does this backend's ``_async`` declare a ``readonly`` parameter?

    Only an explicit parameter counts: today's ``_async(self, func, *args,
    quarantine_exit=False, **kwargs)`` would forward ``readonly`` to ``func``.
    """
    try:
        return "readonly" in inspect.signature(backend._async).parameters
    except (TypeError, ValueError):
        return False


async def _async_readonly(backend, func):
    """``backend._async(func, readonly=True)`` where Task 1's flag exists,
    ``backend._async(func)`` where it does not yet."""
    if _accepts_readonly(backend):
        return await backend._async(func, readonly=True)
    return await backend._async(func)


def _limit() -> int:
    return int(config.settings.max_qa_entities)


def _over_limit(counts: dict[str, int], limit: int) -> str | None:
    total = sum(counts.values())
    if limit > 0 and total > limit:
        spaces = ", ".join(f"{name} {count:,}" for name, count in counts.items())
        return (
            f"{total:,} entities ({spaces}) exceed MAX_QA_ENTITIES={limit:,}; the drawing "
            "was not read. Raise MAX_QA_ENTITIES (0 disables the check) to scan it."
        )
    return None


def _ezdxf_counts(doc) -> dict[str, int]:
    return {name: len(doc.layouts.get(name)) for name in doc.layouts.names_in_taborder()}


async def _read_ezdxf(backend) -> tuple[Snapshot | None, ScanCoverage]:
    limit = _limit()

    def _sync():
        doc = backend._require_doc()
        counts = _ezdxf_counts(doc)
        refusal = _over_limit(counts, limit)
        if refusal is not None:
            return None, counts, refusal
        source = backend._doc_path or "memory:untitled"
        return records_from_doc(doc, source=source), counts, None

    snap, counts, refusal = await _async_readonly(backend, _sync)
    if snap is None:
        return None, ScanCoverage(False, 0, tuple(counts), refusal)
    return snap, ScanCoverage(True, len(snap.records), snap.layouts)


def _com_live_counts(doc) -> dict[str, int]:
    """Entities per layout, in tab order (members measured on AutoCAD 2026)."""
    layouts = doc.Layouts
    rows = []
    for index in range(int(layouts.Count)):
        layout = layouts.Item(index)
        rows.append((int(layout.TabOrder), str(layout.Name), int(layout.Block.Count)))
    return {name: count for _order, name, count in sorted(rows)}


def _com_job(backend, limit: int):
    """Count, then export - one COM job, so nothing of ours runs in between."""
    from backends import com_backend as com

    doc = com._acad_doc()
    counts = backend._wait_out_rejected_call(lambda: _com_live_counts(doc))
    refusal = _over_limit(counts, limit)
    if refusal is not None:
        return None, None, None, counts, refusal
    written, folder, source = _com_export(backend, None)
    return written, folder, source, counts, None


async def _read_com(backend) -> tuple[Snapshot | None, ScanCoverage]:
    written, folder, source, live, refusal = await backend._run(_com_job, backend, _limit())
    if refusal is not None:
        return None, ScanCoverage(False, 0, tuple(live), refusal)
    try:
        snap = await asyncio.to_thread(_read, written, source=source)
    finally:
        shutil.rmtree(folder, ignore_errors=True)
    read = Counter(record.space for record in snap.records)
    differences = [
        f"{name}: AutoCAD holds {live.get(name, 0):,}, the export carried {read.get(name, 0):,}"
        for name in sorted(set(live) | set(read))
        if live.get(name, 0) != read.get(name, 0)
    ]
    if differences:
        reason = "the DXF export does not match the live drawing (" + "; ".join(differences) + ")"
        return snap, ScanCoverage(False, len(snap.records), snap.layouts, reason)
    return snap, ScanCoverage(True, len(snap.records), snap.layouts)


async def _read_all_spaces(backend) -> tuple[Snapshot | None, ScanCoverage]:
    if backend.name == "com":
        return await _read_com(backend)
    return await _read_ezdxf(backend)


async def qa_snapshot(backend, shared: dict) -> tuple[Snapshot | None, ScanCoverage]:
    """The run's one all-spaces read, cached in ``shared["qa_snapshot"]``.

    Returns ``(snapshot, coverage)``. ``snapshot`` is None when nothing was
    read (over ``MAX_QA_ENTITIES``, or the read failed); a live export whose
    counts disagree with AutoCAD's returns what it read with
    ``coverage.complete`` False. A ``DocumentQuarantineError`` propagates and
    is not cached.
    """
    cached = shared.get(QA_SNAPSHOT_KEY)
    if cached is not None:
        return cached
    try:
        result = await _read_all_spaces(backend)
    except DocumentQuarantineError:
        raise
    except Exception as exc:  # noqa: BLE001 - reported as an incomplete scan, never as clean
        reason = f"the drawing could not be read: {type(exc).__name__}: {exc}"
        result = (None, ScanCoverage(False, 0, (), reason))
    shared[QA_SNAPSHOT_KEY] = result
    return result

def _com_infos(backend):
    from backends import com_backend as com

    doc = com._acad_doc()
    layouts = doc.Layouts
    ordered = sorted(
        (int(layouts.Item(i).TabOrder), str(layouts.Item(i).Name), layouts.Item(i).Block)
        for i in range(int(layouts.Count))
    )
    out = []
    failed: list[str] = []
    for _order, name, block in ordered:
        for index in range(int(block.Count)):
            try:

            except Exception as exc:  # noqa: BLE001 - counted and refused below, never skipped

        raise RuntimeError(
            f"{len(failed)} entities could not be read ({'; '.join(failed[:5])}); "
            "the drawing was not read in full"
        )
    return out

class _FailedBuild:
    """A domain index whose build raised, cached so the run reads once."""

    def __init__(self, error: Exception):
        self.error = error


async def shared_index(shared: dict, key: str, build, backend):
    """``shared[key]``, built once per run by ``await build(backend)``.

    A build that raises is cached and re-raised for every later focus of the
    run, so each focus reports ``check_failed`` (``run_critique`` turns the
    exception into that finding) and the drawing is read once. A quarantine
    refusal propagates and is not cached.
    """
    cached = shared.get(key)
    if isinstance(cached, _FailedBuild):
        raise cached.error
    if cached is not None:
        return cached
    try:
        value = await build(backend)
    except DocumentQuarantineError:
        raise
    except Exception as exc:
        shared[key] = _FailedBuild(exc)
        raise
    shared[key] = value
    return value
```

```python
def _all_pairs(points, tol):
    """The oracle: the O(n^2) loop the checks used to run."""
    out = []
    for i in range(len(points)):
        for j in range(i + 1, len(points)):
            if points[i][0] != points[j][0]:
                continue
            d = math.hypot(points[i][1] - points[j][1], points[i][2] - points[j][2])
            if d <= tol:
                out.append((i, j, d))
    return out
```

```python
async def _records(backend: AutoCADBackend, shared: dict, focus: str):
    """The run's snapshot records (every space), plus a ``scan_incomplete``
    issue for ``focus`` when the read did not cover the whole drawing."""
    snap, coverage = await qa_snapshot(backend, shared)
    issues = [] if coverage.complete else [scan_incomplete_issue(focus, coverage)]
    return (snap.records if snap is not None else ()), issues


def scan_incomplete_issue(focus: str, coverage: ScanCoverage) -> Issue:
    """The finding a check reports when its read did not cover the drawing."""
    return Issue(
        severity="error",
        focus=focus,  # type: ignore[arg-type]
        message=f"{focus} could not check the whole drawing: {coverage.reason}",
        detail={"code": SCAN_INCOMPLETE, **coverage.to_dict()},
    )


def _close_pairs(
    points: list[tuple[str, float, float]], tol: float
) -> list[tuple[int, int, float]]:
    """Every pair ``(i, j, d)``, ``i < j``, of points in the same space with
    ``d = hypot(p_i - p_j) <= tol``, sorted by ``(i, j)``.

    A grid hash replaces the all-pairs loop (8.9 s at 5,000 lines on the event
    loop, measured): the cell is the tolerance, so a partner lies in one of
    the nine cells around a point. The cell is widened by one part in a
    million so a float quotient at a cell boundary cannot hide a pair that is
    exactly ``tol`` apart. Sorting by ``(i, j)`` reproduces the order the
    all-pairs loop reported in.
    """
    cell = tol * (1.0 + 1e-6)
    grid: dict[tuple[str, int, int], list[int]] = {}
    keys = []
    for index, (space, x, y) in enumerate(points):
        key = (space, math.floor(x / cell), math.floor(y / cell))
        keys.append(key)
        grid.setdefault(key, []).append(index)
    pairs: list[tuple[int, int, float]] = []
    for index, (space, x, y) in enumerate(points):
        _space, cx, cy = keys[index]
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                for other in grid.get((space, cx + dx, cy + dy), ()):
                    if other <= index:
                        continue
                    d = math.hypot(x - points[other][1], y - points[other][2])
                    if d <= tol:
                        pairs.append((index, other, d))
    pairs.sort()
    return pairs


async def _check_construction_left(backend: AutoCADBackend, shared: dict) -> list[Issue]:
    """No construction/scratch geometry may remain before drawing_finalize.

    Matches any layer whose name contains 'CONST' (CONSTRUCTION, M-CONST-E-N, …)
    so the check is independent of the active layer set — a leftover iso13567
    scaffold on M-CONST-E-N is caught just like a mech one on CONSTRUCTION.
    Every entity of every space is read (``engineering/qa_scan.py``).
    """
    records, issues = await _records(backend, shared, "construction_left")
    leftover = [r for r in records if "CONST" in (r.layer or "").upper()]
    if leftover:
        issues.append(
            Issue(

                focus="construction_left",

                    "layers": sorted({r.layer for r in leftover if r.layer}),
                    "spaces": sorted({r.space for r in leftover}),

    return issues


#: Two LINE ends this close but not equal are an untrimmed corner (mm).
UNTRIMMED_TOL_MM = 0.5


def _untrimmed_issues(records) -> list[Issue]:
    endpoints = [
        (r.handle, r.space, point)
        for r in records
        if r.type == "LINE" and len(r.points) == 2
        for point in r.points
    ]
    pairs = _close_pairs([(space, p[0], p[1]) for _h, space, p in endpoints], UNTRIMMED_TOL_MM)
    issues: list[Issue] = []
    seen: set[frozenset[str]] = set()
    for i, j, d in pairs:
        h1, space, p1 = endpoints[i]
        h2, _space, p2 = endpoints[j]
        if h1 == h2 or d <= 1e-9:
            continue
        key = frozenset((h1, h2))
        if key in seen:
            continue
        seen.add(key)
        issues.append(
            Issue(
                severity="warning",
                focus="untrimmed_corner",
                message=(
                    f"Lines {h1} and {h2} have endpoints {d:.3f} mm apart — "
                    "likely an untrimmed corner (use entity_trim or entity_fillet)."
                ),
                handles=[h1, h2],
                detail={"gap_mm": d, "p1": list(p1), "p2": list(p2), "space": space},
            )
        )
    return issues


async def _check_untrimmed_corner(backend: AutoCADBackend, shared: dict) -> list[Issue]:
    """Two LINE endpoints within ``UNTRIMMED_TOL_MM`` but not exactly equal usually
    means a corner where one line overshoots — TRIM/EXTEND/FILLET was forgotten.
    Ends are compared only within one space; the arithmetic runs off the event loop."""
    records, issues = await _records(backend, shared, "untrimmed_corner")
    return issues + await asyncio.to_thread(_untrimmed_issues, records)


async def _check_duplicate_entities(backend: AutoCADBackend, shared: dict) -> list[Issue]:
    """Two LINEs of one space with identical endpoints (in either direction)."""
    records, issues = await _records(backend, shared, "duplicate_entities")
    seen: dict[tuple, str] = {}
    for r in records:
        if r.type != "LINE" or len(r.points) != 2:
            continue
        (sx, sy), (ex, ey) = r.points
        k1 = (r.space, round(sx, 4), round(sy, 4), round(ex, 4), round(ey, 4))
        k2 = (r.space, k1[3], k1[4], k1[1], k1[2])
        for key in (k1, k2):
            if key in seen:
                issues.append(
                    Issue(
                        severity="warning",
                        focus="duplicate_entities",
                        message=f"Lines {seen[key]} and {r.handle} are duplicates.",
                        handles=[seen[key], r.handle],
                        detail={"space": r.space},
                    )
                )
                break
        else:
            seen[k1] = r.handle
    return issues


def _dim_ref_point(record) -> tuple[float, float] | None:
    """A dimension record's reference point: its text midpoint (the snapshot
    stores ``(defpoint, text midpoint)``, the midpoint falling back to the
    defpoint when the DXF has none), else its box centre."""
    if len(record.points) >= 2:
        return record.points[1]
    if record.points:
        return record.points[0]
    if record.bbox is not None:
        x0, y0, x1, y1 = record.bbox
        return ((x0 + x1) / 2.0, (y0 + y1) / 2.0)
    return None

DIM_OVERLAP_TOL_MM = 5.0

                handles=[h1, h2],
                detail={"space": space},
```

```python
async def _run_check(check, focus: str, backend: AutoCADBackend, shared: dict) -> list[Issue]:
    """One check; a failure becomes an ``error`` issue of that focus, never silence.

    A ``DocumentQuarantineError`` propagates: the document is untrusted and the
    caller must be told so, not handed a finding about it.
    """
    try:
        if getattr(check, "needs_shared", False):
            return await check(backend, shared)
        return await check(backend)
    except DocumentQuarantineError:
        raise
    except Exception as exc:  # noqa: BLE001 - reported as check_failed below
        error = f"{type(exc).__name__}: {exc}"
        return [
            Issue(
                severity="error",
                focus=focus,  # type: ignore[arg-type]
                message=f"The {focus} check failed and checked nothing: {error}",
                detail={"code": CHECK_FAILED, "error": error},
            )
        ]


async def run_critique(
    backend: AutoCADBackend,
    focus: list[CritiqueFocus] | None = None,
    *,
    shared: dict | None = None,
) -> list[Issue]:
    """Dispatch each requested focus check and return aggregated issues.

    ``shared`` is the run's context: the all-spaces QA snapshot
    (``engineering/qa_scan.py``) and the domain indexes (P&ID graph,
    mechanical and architectural indexes, topology) are built once into it.
    Pass the dict the validator used to reuse its read (``drawing_finalize``
    does); a dict is only valid while the drawing does not change. None starts
    a fresh one.

    A check that raises yields one ``error`` issue with
    ``detail["code"] == "check_failed"``; a read that did not cover the whole
    drawing yields ``detail["code"] == "scan_incomplete"``. Neither code is a
    focus. A quarantine refusal propagates.
    """
    foci = list(focus) if focus is not None else list(ALL_CRITIQUE_FOCUSES)
    issues: list[Issue] = []
    if shared is None:
        shared = {}
    for f in foci:
        check = _FOCUS_DISPATCH.get(f)
        if check is None:
            issues.append(
                Issue(
                    severity="warning",
                    focus=f,  # type: ignore[arg-type]
                    message=f"Unknown critique focus '{f}' (closed enum).",
                )
            )
            continue
        issues.extend(await _run_check(check, f, backend, shared))
    return issues
```

```python
class DrawingValidator:
    """8-step pre-completion validator."""

    async def run(
        self,
        backend: AutoCADBackend,
        *,
        expected: dict | None = None,
        shared: dict | None = None,
    ) -> ValidationResult:
        """Run all 8 checks; return aggregate ValidationResult.

        Steps 1, 2, 4 and 5 read the run's all-spaces QA snapshot
        (``engineering/qa_scan.py``): every entity of every space, by its DXF
        type name on both engines. ``shared`` is that run's context - pass the
        dict ``run_critique`` will receive so a finalize reads the drawing
        once. A read that did not cover the drawing is an ``error`` finding
        ``scan_incomplete``; a step that raises is a finding naming the step;
        a quarantine refusal propagates. Nothing is swallowed.
        """

        findings: list[ValidationFinding] = []
        expected = expected or {}
        if shared is None:
            shared = {}
        records: tuple = ()

        snap, coverage = await qa_snapshot(backend, shared)
        if snap is not None:
            records = snap.records
        if not coverage.complete:
            findings.append(
                ValidationFinding(
                    "error",
                    "scan_incomplete",
                    f"The drawing was not read in full: {coverage.reason}",
                    coverage.to_dict(),
                )
            )

        # 1 — fake dimensions
        text_likes = [
            r for r in records if r.type in ("TEXT", "MTEXT") and looks_like_dimension(r.text or "")
        ]
        real_dims = [r for r in records if r.type in REAL_DIMENSION_TYPES]
        if text_likes:
            findings.append(
                ValidationFinding(
                    "warning",
                    "fake_dimension_text",
                    f"{len(text_likes)} TEXT/MTEXT entities look like dimensions "
                    "(Ø, ±, R, etc.) — use dimension_* tools instead",
                    {
                        "sample_handles": [t.handle for t in text_likes[:5]],
                        "real_dim_count": len(real_dims),
                    },
                )
            )

        # 2 — hidden orphans
        hidden = [r for r in records if not r.visible]
        if hidden:
            findings.append(
                ValidationFinding(
                    "error",
                    "hidden_orphans",
                    f"{len(hidden)} entities have visible=False — hide-fallback "
                    "is forbidden, use entity_delete",
                    {
                        "sample_handles": [h.handle for h in hidden[:5]],
                        "spaces": sorted({h.space for h in hidden}),
                    },
                )
            )

        # 3 — linetypes
        try:
            lts = {x.lower() for x in await backend.linetype_list()}
            missing = [lt for lt in ("center", "hidden", "phantom") if lt not in lts]
            if missing:
                findings.append(
                    ValidationFinding(
                        "error",
                        "linetypes_missing",
                        f"Standard linetypes not loaded: {', '.join(missing).upper()}",
                        {"missing": missing},
                    )
                )
        except DocumentQuarantineError:
            raise
        except Exception as exc:
            findings.append(
                ValidationFinding(
                    "warning",
                    "linetype_check_failed",
                    str(exc),
                )
            )

        # 4 — gear views (heuristic, only if expected demands)
        if expected.get("must_have_bore") or expected.get("must_have_keyway"):
            bore_circles = [r for r in records if r.type == "CIRCLE" and r.layer == "GEOMETRY"]
            if expected.get("must_have_bore") and len(bore_circles) < 1:
                findings.append(
                    ValidationFinding(
                        "error",
                        "bore_missing",
                        "Expected bore but no circle on GEOMETRY layer found",
                    )
                )
            # draw_keyed_bore emits the keyway notch as an OPEN 4-point polyline
            # on GEOMETRY — requiring `closed` here made this a permanent
            # false-negative. Match a small (3-6 point) polyline, open or closed.
            keyway_polys = [
                r
                for r in records
                if r.type in ("LWPOLYLINE", "POLYLINE")
                and r.layer == "GEOMETRY"
                and 3 <= len(r.points) <= 6
            ]
            if expected.get("must_have_keyway") and len(keyway_polys) < 1:
                findings.append(
                    ValidationFinding(
                        "warning",
                        "keyway_section_unverified",
                        "Could not confirm keyway via heuristic "
                        "(small 3-6 point polyline on GEOMETRY)",
                    )
                )

        # 5 — title consistency
        helix = expected.get("helix_angle")
        if helix:
            # R27 — scope the scan to title-block text where possible (TITLE /
            # TITLEBLOCK layer); fall back to all TEXT-layer text if no title
            # layer carries any text.
            title_block_texts = [
                r.text or ""
                for r in records
                if r.type in ("TEXT", "MTEXT") and r.layer in _TITLE_LAYERS
            ]
            if title_block_texts:
                title_texts = title_block_texts
            else:
                title_texts = [
                    r.text or ""
                    for r in records
                    if r.type in ("TEXT", "MTEXT") and r.layer == "TEXT"
                ]
            joined = " | ".join(title_texts)
            # R27 — whole-word matching so "SPURIOUS" no longer trips SPUR.
            has_spur = bool(_SPUR_RE.search(joined))
            has_helical = bool(_HELICAL_RE.search(joined))
            if has_spur and has_helical:
                findings.append(
                    ValidationFinding(
                        "error",
                        "title_helical_spur",
                        "Title contains 'HELICAL SPUR GEAR' — invalid; use "
                        "'HELICAL GEAR' or 'SPUR GEAR'",
                    )
                )
            elif has_spur:
                findings.append(
                    ValidationFinding(
                        "error",
                        "title_says_spur_for_helical",
                        f"Title contains SPUR but part is helical (helix_angle={helix})",
                        {"sample": joined[:200]},
                    )
                )

        # 6 — saved to disk
        try:
            info = await backend.drawing_info()
            saved_attr = getattr(info, "saved", None)
            full_path = getattr(info, "full_path", "") or ""
            if not saved_attr:
                findings.append(
                    ValidationFinding(
                        "error",
                        "not_saved",
                        "Drawing has unsaved changes — call drawing_save before finalize",
                        {"full_path": full_path},
                    )
                )
            elif full_path and not Path(full_path).exists():
                findings.append(
                    ValidationFinding(
                        "error",
                        "file_missing",
                        f"Drawing reports saved=True but file not on disk: {full_path}",
                    )
                )
        except DocumentQuarantineError:
            raise
        except Exception as exc:
            findings.append(
                ValidationFinding(
                    "warning",
                    "save_check_failed",
                    str(exc),
                )
            )

        # 7 — screenshot
        try:
            shot = await backend.view_screenshot()
            if not shot:
                findings.append(
                    ValidationFinding(
                        "info",
                        "screenshot_unavailable",
                        "Screenshot returned no bytes (ezdxf needs matplotlib install)",
                    )
                )
        except DocumentQuarantineError:
            raise
        except Exception as exc:
            findings.append(
                ValidationFinding(
                    "info",
                    "screenshot_failed",
                    str(exc),
                )
            )

        # 8 — dwg path returned
        try:
            info = await backend.drawing_info()
            if not getattr(info, "full_path", ""):
                findings.append(
                    ValidationFinding(
                        "warning",
                        "no_path",
                        "drawing_info.full_path is empty",
                    )
                )
        except DocumentQuarantineError:
            raise
        except Exception as exc:
            findings.append(
                ValidationFinding(
                    "warning",
                    "path_check_failed",
                    str(exc),
                )
            )

        summary = {
            sev: sum(1 for f in findings if f.severity == sev)
            for sev in ("error", "warning", "info")
        }
        return ValidationResult(
            ok=summary["error"] == 0,
            findings=findings,
            summary=summary,
        )
```

```python
"""Shared spec-input guards: every refusal names the offending path (B07).

Lifted from ``engineering/mech/part.py``'s private guards so the P&ID, arch,
mechanical and block-definition specs refuse the same inputs the same way:

    equipment[2].scale: must be greater than zero, got -1.0.

``SpecError`` is a ``ValueError``, so every existing ``except ValueError`` (and
the server's refusal mapping) keeps working. A value echoed back in a message
is shortened with ``reprlib`` — a 2 MB string must not become a 2 MB error.
``within_budget`` names the setting that raises its limit, because a count
over a budget is not malformed input, it is input bigger than this server was
configured to draw.

Pure: no backend, no I/O.
"""

from __future__ import annotations

import math
import reprlib
from typing import Any

__all__ = [
    "SpecError",
    "finite",
    "int_in_range",
    "non_negative",
    "positive",
    "short_repr",
    "within_budget",
]

#: The tolerance ``positive`` / ``non_negative`` allow around zero (the value
#: ``engineering/mech/part.py`` has always used).
EPS = 1e-9

_REPR = reprlib.Repr()
_REPR.maxstring = 80
_REPR.maxother = 80
_REPR.maxlong = 40
_REPR.maxlist = 6
_REPR.maxdict = 6


class SpecError(ValueError):
    """A spec value refused before anything is drawn; the message starts with its path."""


def short_repr(value: Any) -> str:
    """``repr(value)``, at most about 80 characters."""
    return _REPR.repr(value)


def finite(value: Any, path: str) -> float:
    """``value`` as a finite float; bools, strings and non-numbers are refused."""
    if isinstance(value, bool) or not isinstance(value, (int, float)):
        raise SpecError(f"{path}: expected a number, got {short_repr(value)}.")
    try:
        number = float(value)
    except OverflowError:  # a 400-digit int is a number, but not a float
        raise SpecError(f"{path}: {short_repr(value)} is not a finite number.") from None
    if not math.isfinite(number):
        raise SpecError(f"{path}: {short_repr(value)} is not a finite number.")
    return number


def positive(value: Any, path: str) -> float:
    """A finite number greater than zero (more than ``EPS``)."""
    number = finite(value, path)
    if number <= EPS:
        raise SpecError(f"{path}: must be greater than zero, got {number}.")
    return number


def non_negative(value: Any, path: str) -> float:
    """A finite number that is not negative; ``-EPS..0`` reads as ``0.0``."""
    number = finite(value, path)
    if number < -EPS:
        raise SpecError(f"{path}: must not be negative, got {number}.")
    return max(number, 0.0)


def int_in_range(value: Any, path: str, lo: int, hi: int) -> int:
    """An ``int`` (never a bool, never a float) with ``lo <= value <= hi``."""
    if isinstance(value, bool) or not isinstance(value, int):
        raise SpecError(f"{path}: expected an integer, got {short_repr(value)}.")
    if not lo <= value <= hi:
        raise SpecError(f"{path}: must be from {lo} to {hi}, got {short_repr(value)}.")
    return value


def within_budget(count: int, limit: int, path: str, setting: str) -> None:
    """Refuse ``count`` above ``limit``, naming ``setting`` (the env var that
    raises the limit). ``limit <= 0`` disables the budget."""
    if limit > 0 and count > limit:
        raise SpecError(
            f"{path}: {count} is over the limit of {limit}; set {setting} higher to allow "
            "more (0 disables the limit)."
        )
```

```python
# engineering/pid/symbols_vessels.py
"""Parametric vessels (ISO 10628-2): size and nozzles come from the caller.

Unlike the fixed-size catalogue, a vessel's block definition depends on its
parameters, so the block name carries an 8-hex hash of the validated parameter
set (spec 4.5): identical parameters share one definition, different ones get
their own. Nozzles become stub lines plus named ports; the default set is
``N1`` top, ``N2`` bottom, ``N3`` left (0.7 up the wall), ``N4`` right (0.3).

A nozzle's port sits ``STUB`` outside the vessel's bounding box on the side it
was asked for; its stub is drawn from the port inward to the point where that
axis-aligned line meets the *drawn* outline (a dished head, a hopper's sloped
wall, a cone roof), so no stub ever floats off the vessel or pierces it.

1.7.0 (B07): every number goes through ``engineering/validation.py`` (NaN and
infinity are refused by path), a column's trays are bounded by
``MAX_VESSEL_TRAYS`` and by a 1 mm spacing and are clipped to the dished heads,
two nozzles may not share one place, and a nozzle whose shell point is half way
or more to the centre is refused. All of it runs in ``validate_vessel_params``,
so a ``pid_from_spec`` dry run refuses exactly what the real run refuses.
"""

from __future__ import annotations

import sys

import config
from engineering.validation import SpecError, finite, int_in_range, positive, within_budget

from .geometry import arc, line, polyline, scan_hits
from .symbols import Port, SymbolSpec, make_spec, params_hash, register, tag_attdefs

VESSELS = ("vertical_vessel", "horizontal_vessel", "tank", "column", "reactor", "hopper")
SIDES = ("top", "bottom", "left", "right")
DEFAULT_NOZZLES = [
    {"name": "N1", "side": "top", "fraction": 0.5},
    {"name": "N2", "side": "bottom", "fraction": 0.5},
    {"name": "N3", "side": "left", "fraction": 0.7},
    {"name": "N4", "side": "right", "fraction": 0.3},
]
_DEFAULTS: dict[str, dict] = {
    "vertical_vessel": {"width": 20.0, "height": 40.0},
    "horizontal_vessel": {"width": 40.0, "height": 20.0},
    "tank": {"width": 24.0, "height": 24.0, "roof": "flat"},
    "column": {"width": 16.0, "height": 60.0, "trays": 0},
    "reactor": {"width": 24.0, "height": 36.0, "jacketed": False},
    "hopper": {"width": 24.0, "height": 24.0},
}
#: Capsule bodies (cylinder with two dished heads) whose axis is vertical.
_VERTICAL_CAPSULES = ("vertical_vessel", "column", "reactor")
PARAMS_SCHEMA = {
    "width": "float > 0 (mm); vertical_vessel/column/reactor need width < height",
    "height": "float > 0 (mm); horizontal_vessel needs height < width; cone tank height > width/6",
    "nozzles": "[{name, side: top|bottom|left|right, fraction: 0..1}]",
    "roof": "tank only: flat | cone",
    "trays": "column only: int >= 0, at most MAX_VESSEL_TRAYS, at least 1 mm apart",
    "jacketed": "reactor only: bool",
}
#: Distance of a nozzle port beyond the vessel's bounding box (symbol-local mm);
#: the stub runs from there in to the drawn outline, so it is at least this long.
STUB = 3.0
#: Gap between a jacketed reactor's wall and its jacket line.
JACKET = 2.0
#: A tray line stops this far inside the drawn shell at both ends.
TRAY_INSET = 1.0
#: Closest two trays may sit (symbol-local mm).
MIN_TRAY_SPACING = 1.0


def validate_vessel_params(symbol: str, params: dict | None) -> dict:
    """Merge ``params`` over the vessel's defaults and validate every key.

    Raises ``ValueError`` (``SpecError``) naming the offending key (and nozzle
    index) before anything is built; the returned dict is the canonical
    parameter set the block name is hashed from. The shell is built here once
    so the tray and nozzle placements a drawing would need are refused here,
    not half way through ``build_vessel``.
    """
    if symbol not in VESSELS:
        raise ValueError(f"unknown vessel {symbol!r}; valid: {', '.join(VESSELS)}")
    merged = dict(_DEFAULTS[symbol])
    merged["nozzles"] = [dict(n) for n in DEFAULT_NOZZLES]
    if symbol == "hopper":
        merged["nozzles"] = [dict(n) for n in DEFAULT_NOZZLES[:2]]
    if params is not None and not isinstance(params, dict):
        raise ValueError(f"{symbol}: params must be a dict")
    for key, value in (params or {}).items():
        if key not in merged:
            raise ValueError(
                f"{symbol}: unknown parameter {key!r}; valid: {', '.join(sorted(merged))}"
            )
        merged[key] = value
    for key in ("width", "height"):
        merged[key] = positive(merged[key], f"{symbol}: params.{key}")
    w, h = merged["width"], merged["height"]
    if symbol in _VERTICAL_CAPSULES and h <= w:
        raise ValueError(
            f"{symbol}: height {h} must exceed width {w} — the dished heads are half-circles "
            "of the width, so a shorter axis folds them through each other"
        )
    if symbol == "horizontal_vessel" and w <= h:
        raise ValueError(
            f"{symbol}: width {w} must exceed height {h} — the dished heads are half-circles "
            "of the height, so a shorter axis folds them through each other"
        )
    if "roof" in merged and merged["roof"] not in ("flat", "cone"):
        raise ValueError(f"{symbol}: roof must be 'flat' or 'cone'")
    if merged.get("roof") == "cone" and h <= w / 6:
        raise ValueError(
            f"{symbol}: height {h} must exceed width/6 ({w / 6}) — the cone roof is that tall"
        )
    if "trays" in merged:
        trays = int_in_range(merged["trays"], f"{symbol}: params.trays", 0, sys.maxsize)
        within_budget(
            trays, config.settings.max_vessel_trays, f"{symbol}: params.trays", "MAX_VESSEL_TRAYS"
        )
        spacing = h / (trays + 1)
        if trays and spacing < MIN_TRAY_SPACING:
            raise SpecError(
                f"{symbol}: params.trays: {trays} trays in a {h:g} mm shell sit {spacing:.3g} mm "
                f"apart; they must be at least {MIN_TRAY_SPACING:g} mm apart"
            )
    if "jacketed" in merged and not isinstance(merged["jacketed"], bool):
        raise ValueError(f"{symbol}: jacketed must be a bool")
    nozzles = merged["nozzles"]
    if not isinstance(nozzles, list):
        raise ValueError(f"{symbol}: nozzles must be a list")
    names: set[str] = set()
    places: dict[tuple[str, float], int] = {}
    clean = []
    for i, n in enumerate(nozzles):
        if not isinstance(n, dict) or not isinstance(n.get("name"), str) or not n["name"]:
            raise ValueError(f"{symbol}: nozzles[{i}].name must be a non-empty string")
        if n["name"] in names:
            raise ValueError(f"{symbol}: nozzles[{i}].name {n['name']!r} is repeated")
        names.add(n["name"])
        for key in n:
            if key not in ("name", "side", "fraction"):
                raise ValueError(
                    f"{symbol}: nozzles[{i}]: unknown key {key!r}; valid: fraction, name, side"
                )
        if n.get("side") not in SIDES:
            raise ValueError(f"{symbol}: nozzles[{i}].side must be one of {', '.join(SIDES)}")
        where = f"{symbol}: nozzles[{i}].fraction"
        # `+ 0.0` folds -0.0 into 0.0: the two hashed to different block names.
        f = finite(n.get("fraction", 0.5), where) + 0.0
        if not 0.0 <= f <= 1.0:
            raise SpecError(f"{where}: must be within 0..1, got {f}")
        place = (n["side"], f)
        if place in places:
            raise SpecError(
                f"{symbol}: nozzles[{i}] sits at {n['side']} fraction {f:g}, where "
                f"nozzles[{places[place]}] already is; two nozzles cannot share one stub"
            )
        places[place] = i
        clean.append({"name": n["name"], "side": n["side"], "fraction": f})
    merged["nozzles"] = clean
    shell = _shell(symbol, merged)
    if symbol == "column":
        _trays(shell, merged)
    _nozzles(symbol, clean, shell, w / 2, h / 2, _jacket(symbol, merged))
    return merged


def _shell(symbol: str, p: dict) -> list[dict]:
    """The drawn outline of a validated vessel, centred on the origin."""
    w, h = p["width"], p["height"]
    half_w, half_h = w / 2, h / 2
    if symbol in _VERTICAL_CAPSULES:
        return _capsule_vertical(w, h)
    if symbol == "horizontal_vessel":
        return _capsule_horizontal(w, h)
    if symbol == "tank":
        roof = w / 6 if p["roof"] == "cone" else 0.0
        shell = [
            polyline(
                [
                    (-half_w, -half_h),
                    (half_w, -half_h),
                    (half_w, half_h - roof),
                    (-half_w, half_h - roof),
                ]
            )
        ]
        if roof:
            shell.append(
                polyline(
                    [(-half_w, half_h - roof), (0, half_h), (half_w, half_h - roof)],
                    closed=False,
                )
            )
        return shell
    # hopper: flat top, walls converging to a narrow outlet
    return [polyline([(-half_w, half_h), (half_w, half_h), (w / 6, -half_h), (-w / 6, -half_h)])]


def _jacket(symbol: str, p: dict) -> float:
    return JACKET if symbol == "reactor" and p["jacketed"] else 0.0


def _trays(shell: list[dict], p: dict) -> list[dict]:
    """A column's tray lines, evenly spaced and clipped to the drawn shell.

    The full width used to be drawn at every height, so a tray inside a dished
    head poked through the outline (backlog W2-T10-05).
    """
    h, trays = p["height"], p["trays"]
    out = []
    for i in range(trays):
        y = -h / 2 + (i + 1) * h / (trays + 1)
        hits = scan_hits(shell, "y", y)
        x0, x1 = min(hits) + TRAY_INSET, max(hits) - TRAY_INSET
        if x1 - x0 <= 0.0:
            raise SpecError(
                f"column: params.trays: tray {i + 1} at y={y:g} falls where the head is "
                f"narrower than {2 * TRAY_INSET:g} mm; use fewer trays"
            )
        out.append(line(x0, y, x1, y))
    return out


def _nozzles(
    symbol: str,
    nozzles: list[dict],
    shell: list[dict],
    half_w: float,
    half_h: float,
    extra: float = 0.0,
) -> tuple[list[dict], list[Port]]:
    """Nozzle stub primitives and ports for a vessel whose outline is ``shell``.

    The port sits ``STUB + extra`` outside the ±half_w × ±half_h box (``extra``
    is a jacketed reactor's jacket, which the stub passes through so the port
    still sits on the symbol's outer edge); the stub starts where the port's
    axis-aligned line meets the outermost point of the drawn outline.
    """
    prims: list[dict] = []
    ports: list[Port] = []
    stub = STUB + extra
    for i, n in enumerate(nozzles):
        side, f = n["side"], n["fraction"]
        if side in ("top", "bottom"):
            x = -half_w + f * 2 * half_w
            hits = scan_hits(shell, "x", x)
            if not hits:
                raise RuntimeError(f"nozzle {n['name']!r}: x={x} meets no outline")
            if side == "top":
                y0, y1, d = max(hits), half_h + stub, 90.0
                gap = half_h - y0
            else:
                y0, y1, d = min(hits), -half_h - stub, 270.0
                gap = y0 + half_h
            _refuse_deep_nozzle(symbol, i, n, gap, half_h)
            prims.append(line(x, y0, x, y1))
            ports.append(Port(n["name"], x, y1, d))
        else:
            y = -half_h + f * 2 * half_h
            hits = scan_hits(shell, "y", y)
            if not hits:
                raise RuntimeError(f"nozzle {n['name']!r}: y={y} meets no outline")
            if side == "left":
                x0, x1, d = min(hits), -half_w - stub, 180.0
                gap = x0 + half_w
            else:
                x0, x1, d = max(hits), half_w + stub, 0.0
                gap = half_w - x0
            _refuse_deep_nozzle(symbol, i, n, gap, half_w)
            prims.append(line(x0, y, x1, y))
            ports.append(Port(n["name"], x1, y, d))
    return prims, ports


def _refuse_deep_nozzle(symbol: str, index: int, nozzle: dict, gap: float, half: float) -> None:
    """Refuse a nozzle whose shell point is half way (or more) to the centre.

    ``gap`` is how far inside its side's bounding-box edge the axis-aligned
    nozzle line meets the drawn shell; ``half`` is the half extent across that
    side. Near a dished head's pole or a hopper's outlet corner the stub would
    start deep inside the box and run along the shell — measured: a ``left``
    nozzle at fraction 0 on a 20 x 40 vertical vessel drew a 13 mm stub from the
    bottom pole (backlog W2-T10-06).
    """
    if gap >= half / 2.0:
        raise SpecError(
            f"{symbol}: nozzles[{index}] ({nozzle['name']!r}) at {nozzle['side']} fraction "
            f"{nozzle['fraction']:g} meets the shell {gap:.3g} mm inside the {nozzle['side']} "
            f"edge, half way or more to the centre ({half:g} mm), so its stub would run along "
            "the shell; move it along that side"
        )


def _capsule_vertical(w: float, h: float) -> list[dict]:
    """Cylinder with dished ends, axis vertical: two walls and two half-circle heads."""
    r = w / 2
    body = h / 2 - r
    return [
        line(-r, -body, -r, body),
        line(r, -body, r, body),
        arc(0, body, r, 0, 180),
        arc(0, -body, r, 180, 360),
    ]


def _capsule_horizontal(w: float, h: float) -> list[dict]:
    r = h / 2
    body = w / 2 - r
    return [
        line(-body, -r, body, -r),
        line(-body, r, body, r),
        arc(body, 0, r, 270, 90),
        arc(-body, 0, r, 90, 270),
    ]


def build_vessel(symbol: str = "vertical_vessel", params: dict | None = None) -> SymbolSpec:
    p = validate_vessel_params(symbol, params)
    w, h = p["width"], p["height"]
    half_w, half_h = w / 2, h / 2
    extra = _jacket(symbol, p)
    shell = _shell(symbol, p)
    prims = list(shell)
    if symbol == "column":
        prims += _trays(shell, p)
    elif extra:
        body = half_h - half_w
        prims += [
            line(-half_w - JACKET, -body, -half_w - JACKET, body),
            line(half_w + JACKET, -body, half_w + JACKET, body),
        ]
    nozzle_prims, ports = _nozzles(symbol, p["nozzles"], shell, half_w, half_h, extra)
    prims += nozzle_prims
    xs = [-half_w - extra, half_w + extra] + [pt.x for pt in ports]
    ys = [-half_h, half_h] + [pt.y for pt in ports]
    box = (min(xs), min(ys), max(xs), max(ys))
    return make_spec(
        f"PID_VESSEL_{symbol.upper()}_{params_hash(p)}",
        "vessel",
        symbol,
        None,
        prims,
        tag_attdefs(box[3]),
        ports,
        "ISO 10628-2:2012 vessels",
        bbox=box,
        parametric=True,
        params=p,
    )


def _vessel_builder(symbol: str):
    def build(params: dict | None = None) -> SymbolSpec:
        return build_vessel(symbol, params)

    return build


for _symbol in VESSELS:
    register(_symbol, "vessel", _vessel_builder(_symbol), params_schema=PARAMS_SCHEMA)
```
